  // Raw Data Embedded exactly as provided
  const RAW_DATA = [
    { block: "Bhojpur", season: "Monsoon", year: 2023, cultivated_hectares: 580, harvest_tonnes: 1392, source: "field register" },
    { block: "Bhojpur", season: "Monsoon", year: 2024, cultivated_hectares: 600, harvest_tonnes: 1500, source: "field register" },

    { block: "Wardha", season: "Monsoon", year: 2024, cultivated_hectares: 400, harvest_tonnes: 1080, source: "field register" }
  ];

  // Process data by Block
  const blocksMap = {};
  RAW_DATA.forEach(row => {
    if (!blocksMap[row.block]) {
      blocksMap[row.block] = { block: row.block, source: row.source };
    }
    if (row.year === 2023) {
      blocksMap[row.block].area23 = row.cultivated_hectares;
      blocksMap[row.block].harvest23 = row.harvest_tonnes;
      blocksMap[row.block].yield23 = row.harvest_tonnes / row.cultivated_hectares;
    } else if (row.year === 2024) {
      blocksMap[row.block].area24 = row.cultivated_hectares;
      blocksMap[row.block].harvest24 = row.harvest_tonnes;
      blocksMap[row.block].yield24 = row.harvest_tonnes / row.cultivated_hectares;
    }
  });

  let processedBlocks = Object.values(blocksMap).map(b => {
    const yieldDelta = b.yield24 - b.yield23;
    const yieldGrowthPct = ((b.yield24 - b.yield23) / b.yield23) * 100;
    return {
      ...b,
      yieldDelta,
      yieldGrowthPct
    };
  });

  // Sort by 2024 yield descending (Tonnes per hectare highest first)
  processedBlocks.sort((a, b) => b.yield24 - a.yield24);
  processedBlocks.forEach((b, idx) => b.rank = idx + 1);

  // Aggregates
  const totalArea23 = processedBlocks.reduce((acc, b) => acc + b.area23, 0);
  const totalHarvest23 = processedBlocks.reduce((acc, b) => acc + b.harvest23, 0);
  const avgYield23 = totalHarvest23 / totalArea23;

  const totalArea24 = processedBlocks.reduce((acc, b) => acc + b.area24, 0);
  const totalHarvest24 = processedBlocks.reduce((acc, b) => acc + b.harvest24, 0);
  const avgYield24 = totalHarvest24 / totalArea24;

  const totalYieldDelta = avgYield24 - avgYield23;
  const totalYieldGrowthPct = (totalYieldDelta / avgYield23) * 100;

  // Table Logic: Sorting and Filtering
  let sortCol = "yield24";
  let sortAsc = false;
  let searchTerm = "";

  function renderTable() {
    const tbody = document.getElementById("tableBody");
    
    // Filter
    let filtered = processedBlocks.filter(item => {
      const matchSearch = item.block.toLowerCase().includes(searchTerm) || 
                           item.source.toLowerCase().includes(searchTerm);
      return matchSearch;
    });

    // Sort
    filtered.sort((a, b) => {
      let valA = a[sortCol];
      let valB = b[sortCol];
      if (typeof valA === "string") {
        return sortAsc ? valA.localeCompare(valB) : valB.localeCompare(valA);
      }
      return sortAsc ? valA - valB : valB - valA;
    });

    // Update Th Classes
    document.querySelectorAll("#dataTable th").forEach(th => {
      th.classList.remove("sorted-asc", "sorted-desc");
      if (th.getAttribute("data-col") === sortCol) {
        th.classList.add(sortAsc ? "sorted-asc" : "sorted-desc");
      }
    });

    if (filtered.length === 0) {
      tbody.innerHTML = `<tr><td colspan="11" style="text-align:center; padding: 24px; color: var(--text-muted);">No matching blocks found.</td></tr>`;
      return;
    }

    tbody.innerHTML = filtered.map(b => `
      <tr>
        <td><span class="badge-rank">${b.rank}</span></td>
        <td style="font-weight: 600;">${b.block}</td>
        <td class="num">${b.area23.toLocaleString()}</td>
        <td class="num">${b.harvest23.toLocaleString()}</td>
        <td class="num">${b.yield23.toFixed(2)}</td>
        <td class="num">${b.area24.toLocaleString()}</td>
        <td class="num">${b.harvest24.toLocaleString()}</td>
        <td class="num" style="font-weight: 700; color: #047857;">${b.yield24.toFixed(2)}</td>
        <td class="num" style="color: #059669; font-weight: 600;">+${b.yieldDelta.toFixed(2)}</td>
        <td class="num" style="color: #059669; font-weight: 600;">+${b.yieldGrowthPct.toFixed(2)}%</td>
        <td><span class="source-pill">${b.source}</span></td>
      </tr>
    `).join("");
  }

  // Setup Table Sorting Event
  document.querySelectorAll("#dataTable th").forEach(th => {
    th.addEventListener("click", () => {
      const col = th.getAttribute("data-col");
      if (sortCol === col) {
        sortAsc = !sortAsc;
      } else {
        sortCol = col;
        sortAsc = col === "block" || col === "source" || col === "rank";
      }
      renderTable();
    });
