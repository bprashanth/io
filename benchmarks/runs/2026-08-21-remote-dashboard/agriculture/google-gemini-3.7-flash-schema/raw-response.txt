  <script>
    // DATA PLACEHOLDER
    const DATA = __DATA__;

    // State Variables
    let selectedSeason = "";
    let processedBlocks = [];
    let filteredBlocks = [];
    let currentSort = { column: 'yield_2024', asc: false };

    // Formatters
    const numFmt = new Intl.NumberFormat('en-US', { minimumFractionDigits: 0, maximumFractionDigits: 2 });
    const intFmt = new Intl.NumberFormat('en-US', { maximumFractionDigits: 0 });

    function init() {
      if (!DATA || DATA.length === 0) {
        document.getElementById('tableBody').innerHTML = '<tr><td colspan="9" style="text-align:center; padding: 24px;">No data available.</td></tr>';
        return;
      }

      // Populate seasons dropdown
      const seasons = Array.from(new Set(DATA.map(d => d.season).filter(Boolean)));
      const seasonSelect = document.getElementById('seasonSelect');
      seasonSelect.innerHTML = '';
      
      // Look for "Monsoon" or "Kharif" as default priority
      let defaultSeason = seasons[0];
      const monsoonMatch = seasons.find(s => s.toLowerCase().includes('monsoon') || s.toLowerCase().includes('kharif'));
      if (monsoonMatch) defaultSeason = monsoonMatch;

      seasons.forEach(s => {
        const opt = document.createElement('option');
        opt.value = s;
        opt.textContent = s;
        if (s === defaultSeason) opt.selected = true;
        seasonSelect.appendChild(opt);
      });

      selectedSeason = seasonSelect.value;
      processData();
    }

    function handleSeasonChange() {
      selectedSeason = document.getElementById('seasonSelect').value;
      processData();
    }

    function processData() {
      const seasonRows = DATA.filter(d => d.season === selectedSeason);

      // Collect sources
      const sources = Array.from(new Set(seasonRows.map(d => d.source).filter(Boolean)));
      document.getElementById('sourceDisplay').textContent = sources.length ? sources.join(', ') : 'Not specified';

      // Group by block
      const blockMap = {};

      seasonRows.forEach(row => {
        const b = row.block;
        if (!blockMap[b]) {
          blockMap[b] = {
            block: b,
            source: row.source || 'N/A',
            area_2024: 0,
            harvest_2024: 0,
            area_2023: 0,
            harvest_2023: 0,
            has_2024: false,
            has_2023: false
          };
        }
        if (row.year === 2024) {
          blockMap[b].area_2024 += (Number(row.cultivated_hectares) || 0);
          blockMap[b].harvest_2024 += (Number(row.harvest_tonnes) || 0);
          blockMap[b].has_2024 = true;
        } else if (row.year === 2023) {
          blockMap[b].area_2023 += (Number(row.cultivated_hectares) || 0);
          blockMap[b].harvest_2023 += (Number(row.harvest_tonnes) || 0);
          blockMap[b].has_2023 = true;
        }
        if (row.source) blockMap[b].source = row.source;
      });

      processedBlocks = Object.values(blockMap).map(b => {
        const yield_2024 = b.area_2024 > 0 ? (b.harvest_tonnes_2024 = b.harvest_2024) / b.area_2024 : 0;
        const yield_2023 = b.area_2023 > 0 ? b.harvest_2023 / b.area_2023 : 0;
        
        let yield_change_pct = null;
        if (b.has_2023 && b.has_2024 && yield_2023 > 0) {
          yield_change_pct = ((yield_2024 - yield_2023) / yield_2023) * 100;
        }

        return {
          ...b,
          yield_2024: yield_2024,
          yield_2023: yield_2023,
          yield_change_pct: yield_change_pct
        };
      });

      // Default sort: yield_2024 descending
      currentSort = { column: 'yield_2024', asc: false };
      sortProcessedData();
      
      updateKPIs();
      renderCharts();
      renderTable();
    }

    function updateKPIs() {
      // 2024 totals
      let totalArea24 = 0, totalHarvest24 = 0;
      let totalArea23 = 0, totalHarvest23 = 0;
      let topBlock = null;

      processedBlocks.forEach(b => {
        totalArea24 += b.area_2024;
        totalHarvest24 += b.harvest_2024;
        totalArea23 += b.area_2023;
        totalHarvest23 += b.harvest_2023;

        if (b.has_2024) {
          if (!topBlock || b.yield_2024 > topBlock.yield_2024) {
            topBlock = b;
          }
        }
      });

      const avgYield24 = totalArea24 > 0 ? totalHarvest24 / totalArea24 : 0;
      const avgYield23 = totalArea23 > 0 ? totalHarvest23 / totalArea23 : 0;

      // Yield KPI
      document.getElementById('kpi-2024-yield').textContent = numFmt.format(avgYield24);
      document.getElementById('kpi-2023-yield').textContent = numFmt.format(avgYield23);
      const yieldTrendEl = document.getElementById('kpi-yield-trend');
      if (avgYield23 > 0) {
        const pct = ((avgYield24 - avgYield23) / avgYield23) * 100;
        const sign = pct >= 0 ? '+' : '';
        yieldTrendEl.textContent = `${sign}${pct.toFixed(1)}%`;
        yieldTrendEl.className = `trend-badge ${pct >= 0 ? 'trend-up' : 'trend-down'}`;
      } else {
        yieldTrendEl.textContent = 'N/A';
        yieldTrendEl.className = 'trend-badge';
      }

      // Harvest KPI
      document.getElementById('kpi-2024-harvest').textContent = intFmt.format(totalHarvest24);
      document.getElementById('kpi-2023-harvest').textContent = intFmt.format(totalHarvest23);
      const harvestTrendEl = document.getElementById('kpi-harvest-trend');
      if (totalHarvest23 > 0) {
        const pct = ((totalHarvest24 - totalHarvest23) / totalHarvest23) * 100;
        const sign = pct >= 0 ? '+' : '';
        harvestTrendEl.textContent = `${sign}${pct.toFixed(1)}%`;
        harvestTrendEl.className = `trend-badge ${pct >= 0 ? 'trend-up' : 'trend-down'}`;
      } else {
        harvestTrendEl.textContent = 'N/A';
        harvestTrendEl.className = 'trend-badge';
      }

      // Area KPI
      document.getElementById('kpi-2024-area').textContent = intFmt.format(totalArea24);
      document.getElementById('kpi-2023-area').textContent = intFmt.format(totalArea23);
      const areaTrendEl = document.getElementById('kpi-area-trend');
      if (totalArea23 > 0) {
        const pct = ((totalArea24 - totalArea23) / totalArea23) * 100;
        const sign = pct >= 0 ? '+' : '';
        areaTrendEl.textContent = `${sign}${pct.toFixed(1)}%`;
        areaTrendEl.className = `trend-badge ${pct >= 0 ? 'trend-up' : 'trend-down'}`;
      } else {
        areaTrendEl.textContent = 'N/A';
        areaTrendEl.className = 'trend-badge';
      }

      // Top Block KPI
      if (topBlock) {
        document.getElementById('kpi-top-block').textContent = topBlock.block;
        document.getElementById('kpi-top-block-yield').textContent = `${numFmt.format(topBlock.yield_2024)} t/ha`;
      } else {
        document.getElementById('kpi-top-block').textContent = '-';
        document.getElementById('kpi-top-block-yield').textContent = '-';
      }
    }

    function sortProcessedData() {
      const col = currentSort.column;
      const asc = currentSort.asc;
      processedBlocks.sort((a, b) => {
        let vA = a[col];
        let vB = b[col];
        if (vA === null || vA === undefined) return 1;
        if (vB === null || vB === undefined) return -1;
        if (typeof vA === 'string') {
          return asc ? vA.localeCompare(vB) : vB.localeCompare(vA);
        }
        return asc ? vA - vB : vB - vA;
      });
    }

    function sortTable(column) {
      if (currentSort.column === column) {
        currentSort.asc = !currentSort.asc;
      } else {
        currentSort.column = column;
        currentSort.asc = column === 'block' || column === 'source';
      }
      
      // Update icons
      const headers = ['block', 'yield_2024', 'yield_2023', 'yield_change_pct', 'harvest_2024', 'area_2024', 'harvest_2023', 'area_2023', 'source'];
      headers.forEach(h => {
        const el = document.getElementById(`sort-${h}`);
        if (el) {
          if (h === currentSort.column) {
            el.textContent = currentSort.asc ? '▲' : '▼';
          } else {
            el.textContent = '⇅';
          }
        }
      });

      sortProcessedData();
      renderTable();
    }

    function handleTableSearch(val) {
      const term = val.toLowerCase().trim();
      if (!term) {
        filteredBlocks = [...processedBlocks];
      } else {
        filteredBlocks = processedBlocks.filter(b => b.block.toLowerCase().includes(term));
      }
      renderTableRows();
    }

    function renderTable() {
      const searchVal = document.getElementById('tableFilter').value;
      handleTableSearch(searchVal);
    }

    function renderTableRows() {
      const tbody = document.getElementById('tableBody');
      if (filteredBlocks.length === 0) {
        tbody.innerHTML = '<tr><td colspan="9" style="text-align: center; color: var(--text-muted); padding: 20px;">No matching records found</td></tr>';
        return;
      }

      tbody.innerHTML = filteredBlocks.map(b => {
        let changeCell = '<span style="color: var(--text-muted);">-</span>';
        if (b.yield_change_pct !== null) {
          const isPos = b.yield_change_pct >= 0;
          const sign = isPos ? '+' : '';
          const col = isPos ? 'var(--accent-positive)' : 'var(--accent-negative)';
          changeCell = `<span style="color: ${col}; font-weight: 600;">${sign}${b.yield_change_pct.toFixed(2)}%</span>`;
        }

        return `
          <tr>
            <td style="font-weight: 600;">${escapeHtml(b.block)}</td>
            <td class="num-col" style="color: var(--accent-2024); font-weight: 600;">${b.has_2024 ? numFmt.format(b.yield_2024) : '-'}</td>
            <td class="num-col" style="color: var(--accent-2023);">${b.has_2023 ? numFmt.format(b.yield_2023) : '-'}</td>
            <td class="num-col">${changeCell}</td>
            <td class="num-col">${b.has_2024 ? intFmt.format(b.harvest_2024) : '-'}</td>
            <td class="num-col">${b.has_2024 ? intFmt.format(b.area_2024) : '-'}</td>
            <td class="num-col">${b.has_2023 ? intFmt.format(b.harvest_2023) : '-'}</td>
            <td class="num-col">${b.has_2023 ? intFmt.format(b.area_2023) : '-'}</td>
            <td style="color: var(--text-secondary); font-size: 12px;">${escapeHtml(b.source)}</td>
          </tr>
        `;
      }).join('');
    }

    function escapeHtml(str) {
      if (!str) return '';
      return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
    }

    /* CHART RENDERING (Inline SVG) */
    function renderCharts() {
      renderYieldBarChart();
      renderHarvestAreaChart();
    }

    function renderYieldBarChart() {
      const container = document.getElementById('yieldChartContainer');
      const w = container.clientWidth || 650;
      const h = container.clientHeight || 320;
      const margin = { top: 20, right: 20, bottom: 45, left: 45 };
      const width = w - margin.left - margin.right;
      const height = h - margin.top - margin.bottom;

      // Top 10-15 blocks sorted by 2024 yield desc for chart clarity
      const sorted = [...processedBlocks].sort((a, b) => b.yield_2024 - a.yield_2024).slice(0, 12);

      if (sorted.length === 0) {
        container.innerHTML = '<div style="display:flex;height:100%;align-items:center;justify-content:center;color:var(--text-muted);">No data available</div>';
        return;
      }

      let maxYield = 0;
      sorted.forEach(b => {
        if (b.yield_2024 > maxYield) maxYield = b.yield_2024;
        if (b.yield_2023 > maxYield) maxYield = b.yield_2023;
      });
      maxYield = maxYield * 1.15 || 1;

      // Grouped Bar Math
      const numGroups = sorted.length;
      const groupWidth = width / numGroups;
      const barWidth = Math.min(24, Math.max(6, (groupWidth - 10) / 2));

      // Y Axis Ticks
      const yTicksCount = 5;
      let yTicksHtml = '';
      for (let i = 0; i <= yTicksCount; i++) {
        const val = (maxYield / yTicksCount) * i;
        const yPos = height - (val / maxYield) * height;
        yTicksHtml += `
          <line x1="0" y1="${yPos}" x2="${width}" y2="${yPos}" class="grid-line" />
          <text x="-8" y="${yPos + 4}" text-anchor="end" fill="var(--text-secondary)" font-size="10">${val.toFixed(1)}</text>
        `;
      }

      // Bars & X labels
      let barsHtml = '';
      sorted.forEach((d, idx) => {
        const groupCenterX = idx * groupWidth + groupWidth / 2;
        const h24 = d.has_2024 ? (d.yield_2024 / maxYield) * height : 0;
        const h23 = d.has_2023 ? (d.yield_2023 / maxYield) * height : 0;

        const x24 = groupCenterX - barWidth - 1;
        const y24 = height - h24;

        const x23 = groupCenterX + 1;
        const y23 = height - h23;

        // Label truncating
        const displayLabel = d.block.length > 8 ? d.block.substring(0, 7) + '…' : d.block;

        barsHtml += `
          <g class="bar-group" data-block="${escapeHtml(d.block)}" data-y24="${d.yield_2024.toFixed(2)}" data-y23="${d.yield_2023.toFixed(2)}">
            <rect x="${x24}" y="${y24}" width="${barWidth}" height="${h24}" fill="var(--accent-2024)" rx="2" style="transition: opacity 0.2s;" />
            <rect x="${x23}" y="${y23}" width="${barWidth}" height="${h23}" fill="var(--accent-2023)" rx="2" style="transition: opacity 0.2s;" />
            <text x="${groupCenterX}" y="${height + 16}" text-anchor="middle" fill="var(--text-secondary)" font-size="11" transform="rotate(0 ${groupCenterX} ${height+16})">${escapeHtml(displayLabel)}</text>
          </g>
        `;
      });

      const svg = `
        <svg width="100%" height="100%" viewBox="0 0 ${w} ${h}" preserveAspectRatio="xMidYMid meet">
          <g transform="translate(${margin.left},${margin.top})">
            ${yTicksHtml}
            <line x1="0" y1="${height}" x2="${width}" y2="${height}" stroke="var(--border-color)" stroke-width="1.5" />
            <line x1="0" y1="0" x2="0" y2="${height}" stroke="var(--border-color)" stroke-width="1.5" />
            ${barsHtml}
          </g>
        </svg>
      `;

      container.innerHTML = svg;
      attachBarTooltips(container, 'Yield (t/ha)');
    }

    function renderHarvestAreaChart() {
      const container = document.getElementById('harvestAreaChartContainer');
      const w = container.clientWidth || 650;
      const h = container.clientHeight || 320;
      const margin = { top: 20, right: 45, bottom: 45, left: 45 };
      const width = w - margin.left - margin.right;
      const height = h - margin.top - margin.bottom;

      // Top blocks sorted by 2024 Harvest
      const sorted = [...processedBlocks].sort((a, b) => b.harvest_2024 - a.harvest_2024).slice(0, 12);

      if (sorted.length === 0) {
        container.innerHTML = '<div style="display:flex;height:100%;align-items:center;justify-content:center;color:var(--text-muted);">No data available</div>';
        return;
      }

      let maxHarvest = Math.max(...sorted.map(d => d.harvest_2024), 1) * 1.15;
      let maxArea = Math.max(...sorted.map(d => d.area_2024), 1) * 1.15;

      const numGroups = sorted.length;
      const groupWidth = width / numGroups;
      const barWidth = Math.min(22, Math.max(6, (groupWidth - 12) / 2));

      // Y Left Axis (Harvest - t)
      let yTicksHtml = '';
      const yTicksCount = 5;
      for (let i = 0; i <= yTicksCount; i++) {
        const valH = (maxHarvest / yTicksCount) * i;
        const valA = (maxArea / yTicksCount) * i;
        const yPos = height - (i / yTicksCount) * height;
        yTicksHtml += `
          <line x1="0" y1="${yPos}" x2="${width}" y2="${yPos}" class="grid-line" />
          <text x="-6" y="${yPos + 4}" text-anchor="end" fill="#38bdf8" font-size="10">${intFmt.format(valH)}</text>
          <text x="${width + 6}" y="${yPos + 4}" text-anchor="start" fill="#f59e0b" font-size="10">${intFmt.format(valA)}</text>
        `;
      }

      let elementsHtml = '';
      sorted.forEach((d, idx) => {
        const groupCenterX = idx * groupWidth + groupWidth / 2;
        const hHarvest = (d.harvest_2024 / maxHarvest) * height;
        const hArea = (d.area_2024 / maxArea) * height;

        const xH = groupCenterX - barWidth - 1;
        const yH = height - hHarvest;

        const xA = groupCenterX + 1;
        const yA = height - hArea;

        const displayLabel = d.block.length > 8 ? d.block.substring(0, 7) + '…' : d.block;

        elementsHtml += `
          <g class="dual-bar-group" data-block="${escapeHtml(d.block)}" data-harvest="${d.harvest_2024}" data-area="${d.area_2024}">
            <rect x="${xH}" y="${yH}" width="${barWidth}" height="${hHarvest}" fill="#38bdf8" rx="2" />
            <rect x="${xA}" y="${yA}" width="${barWidth}" height="${hArea}" fill="#f59e0b" rx="2" />
            <text x="${groupCenterX}" y="${height + 16}" text-anchor="middle" fill="var(--text-secondary)" font-size="11">${escapeHtml(displayLabel)}</text>
          </g>
        `;
      });

      const svg = `
        <svg width="100%" height="100%" viewBox="0 0 ${w} ${h}" preserveAspectRatio="xMidYMid meet">
          <g transform="translate(${margin.left},${margin.top})">
            ${yTicksHtml}
            <line x1="0" y1="${height}" x2="${width}" y2="${height}" stroke="var(--border-color)" stroke-width="1.5" />
            <line x1="0" y1="0" x2="0" y2="${height}" stroke="#38bdf8" stroke-width="1" />
            <line x1="${width}" y1="0" x2="${width}" y2="${height}" stroke="#f59e0b" stroke-width="1" />
            ${elementsHtml}
          </g>
        </svg>
      `;

      container.innerHTML = svg;
      attachDualBarTooltips(container);
    }

    function attachBarTooltips(container, unit) {
      const tooltip = document.getElementById('chartTooltip');
      const groups = container.querySelectorAll('.bar-group');

      groups.forEach(g => {
        g.addEventListener('mouseenter', (e) => {
          const b = g.getAttribute('data-block');
          const y24 = g.getAttribute('data-y24');
          const y23 = g.getAttribute('data-y23');
          tooltip.innerHTML = `
            <strong>${b}</strong><br/>
            <span style="color:var(--accent-2024);">● 2024: ${y24} ${unit}</span><br/>
            <span style="color:var(--accent-2023);">● 2023: ${y23} ${unit}</span>
          `;
          tooltip.style.display = 'block';
        });

        g.addEventListener('mousemove', (e) => {
          tooltip.style.left = (e.pageX + 12) + 'px';
          tooltip.style.top = (e.pageY - 28) + 'px';
        });

        g.addEventListener('mouseleave', () => {
          tooltip.style.display = 'none';
        });
      });
    }

    function attachDualBarTooltips(container) {
      const tooltip = document.getElementById('chartTooltip');
      const groups = container.querySelectorAll('.dual-bar-group');

      groups.forEach(g => {
        g.addEventListener('mouseenter', (e) => {
          const b = g.getAttribute('data-block');
          const harvest = numFmt.format(g.getAttribute('data-harvest'));
          const area = numFmt.format(g.getAttribute('data-area'));
          tooltip.innerHTML = `
            <strong>${b} (2024)</strong><br/>
            <span style="color:#38bdf8;">● Harvest: ${harvest} tonnes</span><br/>
            <span style="color:#f59e0b;">● Area: ${area} ha</span>
          `;
          tooltip.style.display = 'block';
        });

        g.addEventListener('mousemove', (e) => {
          tooltip.style.left = (e.pageX + 12) + 'px';
          tooltip.style.top = (e.pageY - 28) + 'px';
        });

        g.addEventListener('mouseleave', () => {
          tooltip.style.display = 'none';
        });
      });
    }

    // CSV Download Builder
    function downloadCSV() {
      if (!processedBlocks || processedBlocks.length === 0) return;

      const headers = [
        "Block",
        "Season",
        "2024 Yield (tonnes/ha)",
        "2023 Yield (tonnes/ha)",
        "YoY Yield Change (%)",
        "2024 Harvest (tonnes)",
        "2024 Cultivated Area (ha)",
        "2023 Harvest (tonnes)",
        "2023 Cultivated Area (ha)",
        "Source"
      ];

      const rows = processedBlocks.map(b => [
        `"${(b.block || '').replace(/"/g, '""')}"`,
        `"${(selectedSeason || '').replace(/"/g, '""')}"`,
        b.has_2024 ? b.yield_2024.toFixed(4) : "",
        b.has_2023 ? b.yield_2023.toFixed(4) : "",
        b.yield_change_pct !== null ? b.yield_change_pct.toFixed(2) : "",
        b.has_2024 ? b.harvest_2024 : "",
        b.has_2024 ? b.area_2024 : "",
        b.has_2023 ? b.harvest_2023 : "",
        b.has_2023 ? b.area_2023 : "",
        `"${(b.source || '').replace(/"/g, '""')}"`
      ]);

      const csvContent = "data:text/csv;charset=utf-8," + [headers.join(","), ...rows.map(e => e.join(","))].join("\n");
      const encodedUri = encodeURI(csvContent);
      const link = document.createElement("a");
      link.setAttribute("href", encodedUri);
      link.setAttribute("download", `Crop_Yield_${selectedSeason.replace(/\s+/g, '_')}_2024_vs_2023.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }
