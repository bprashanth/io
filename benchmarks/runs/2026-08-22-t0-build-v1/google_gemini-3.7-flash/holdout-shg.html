How this was computed · 6 rows
SELECT "Block" AS block, COUNT("Loan ID") AS total_loans, SUM("Loan Amount (Rs)") AS total_disbursed, SUM("Amount Repaid (Rs)") AS total_repaid, ROUND(SUM("Amount Repaid (Rs)") * 100.0 / NULLIF(SUM("Loan Amount (Rs)"), 0), 1) AS recovery_pct, SUM(CASE WHEN "Repayment Status" = 'Overdue' THEN 1 ELSE 0 END) AS overdue_loans, SUM(CASE WHEN "Repayment Status" = 'Overdue' THEN "Loan Amount (Rs)" ELSE 0 END) AS overdue_principal FROM "shg_loan_register" GROUP BY 1 ORDER BY total_disbursed DESC
