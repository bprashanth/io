Average shuttle run improvement by location

How this was computed · 20 rows
SELECT "Location", AVG("end_shuttle" - "base_shuttle") AS avg_improvement FROM (SELECT b."Location", b."Shuttle run sec" AS base_shuttle, e."Shuttle run sec" AS end_shuttle FROM "baseline" b JOIN "endline" e USING ("Child ID")) GROUP BY "Location" ORDER BY avg_improvement DESC LIMIT 20

Average sit‑ups improvement by coach

How this was computed · 15 rows
SELECT "Coach", AVG("end_sit" - "base_sit") AS avg_improvement FROM (SELECT b."Coach", b."Sit ups" AS base_sit, e."Sit ups" AS end_sit FROM "baseline" b JOIN "endline" e USING ("Child ID")) GROUP BY "Coach" ORDER BY avg_improvement DESC LIMIT 20
