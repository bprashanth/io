Avg Shuttle Run Improvement (Seconds)

No rows to chart.

How this was computed · 2 rows
WITH combined AS (
  SELECT "Child ID", "Gender", "Shuttle run sec" AS baseline_val, NULL AS endline_val FROM "baseline"
  UNION ALL
  SELECT "Child ID", "Gender", NULL AS baseline_val, "Shuttle run sec" AS endline_val FROM "endline"
),
matched AS (
  SELECT 
    c1."Gender",
    c1.baseline_val,
    c2.endline_val,
    (c1.baseline_val - c2.endline_val) AS improvement_sec
  FROM combined c1
  JOIN combined c2 ON c1."Child ID" = c2."Child ID" AND c1."Gender" = c2."Gender"
  WHERE c1.baseline_val IS NOT NULL AND c2.endline_val IS NOT NULL
)
SELECT "Gender", AVG(improvement_sec) AS avg_improvement FROM matched GROUP BY 1 ORDER BY 2 DESC

Avg Sit Ups Improvement (Count)

How this was computed · 2 rows
WITH combined AS (
  SELECT "Child ID", "Gender", "Sit ups" AS baseline_val, NULL AS endline_val FROM "baseline"
  UNION ALL
  SELECT "Child ID", "Gender", NULL AS baseline_val, "Sit ups" AS endline_val FROM "endline"
),
matched AS (
  SELECT 
    c1."Gender",
    c1.baseline_val,
    c2.endline_val,
    (c2.endline_val - c1.baseline_val) AS improvement_count
  FROM combined c1
  JOIN combined c2 ON c1."Child ID" = c2."Child ID" AND c1."Gender" = c2."Gender"
  WHERE c1.baseline_val IS NOT NULL AND c2.endline_val IS NOT NULL
)
SELECT "Gender", AVG(improvement_count) AS avg_improvement FROM matched GROUP BY 1 ORDER BY 2 DESC
