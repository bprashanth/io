How this was computed · 3 rows
SELECT 'Total Current Stock' AS metric, COALESCE(SUM(CASE WHEN d1.device_type = 'Braille kit' THEN d1.balance_after WHEN d1.device_type = 'Hearing aid' THEN d1.balance_after WHEN d1.device_type = 'Tricycle' THEN d1.balance_after WHEN d1.device_type = 'Wheelchair' THEN d1.balance_after WHEN d1.device_type = 'Crutches' THEN d1.balance_after WHEN d1.device_type = 'Walking stick' THEN d1.balance_after WHEN d1.device_type = 'Calipers' THEN d1.balance_after END), 0) AS value FROM device_inventory d1 INNER JOIN (SELECT device_type, MAX(date) as max_date FROM device_inventory GROUP BY device_type) d2 ON d1.device_type = d2.device_type AND d1.date = d2.max_date UNION ALL SELECT 'Total Beneficiaries Served' AS metric, COUNT(*) AS value FROM beneficiary_master WHERE "Assistive Device Issued" IS NOT NULL AND "Assistive Device Issued" != '' UNION ALL SELECT 'Total Registered' AS metric, COUNT(*) AS value FROM beneficiary_master
